n = int(input())
a = list(map(int, input().split()))
 
cur = a[0]
pos = 0
ans = float('inf')
for i in range(1,n):
    if a[i] < cur:
        cur = a[i]
        pos = i
        ans = float('inf')
 
    elif a[i] == cur:
        ans = min(ans, i - pos)
        pos = i
    
    # print(cur, pos, ans)
 
print(ans)
